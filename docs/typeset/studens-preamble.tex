% ---------------------------------------------------------------------------
% Palette.
%
% HOMOGENEOUS WITH THE INSTITUTIONS, by François's decision of 2026-09-10.
% An earlier draft argued for a deliberately non-blue identity so the product
% could never read as an official university product. Overruled: the platform
% should sit visually alongside the institutions it serves, and a per
% institution theme is planned (see the theming section).
%
% The primary is #1B4A8F, which the shipped app already uses, so the code and
% this document now agree rather than proposing different things. The sky
% accent is #5EB3E4, taken directly from the UCLouvain mark. Gold with navy is
% the ordinary academic pairing and carries attention without clashing.
%
% TWO COLOURS ARE FUNCTIONAL, NOT DECORATIVE, and they are the reason the
% others are constrained: the attributed path is teal-green and the anonymous
% path is violet, everywhere, at every size. Both are chosen to be
% distinguishable from the blue chrome AND from each other, because the whole
% point is that a reader can never confuse the two paths. They are therefore
% the one part of the palette a theme may not touch.
% ---------------------------------------------------------------------------

  tiny/.style={font=\sffamily\tiny\color{muted}},
  namedchip/.style={draw=named,fill=namedlt,rounded corners=6pt,
                    font=\sffamily\tiny\bfseries\color{named},inner sep=2pt},
  anonchip/.style={draw=anon,fill=anonlt,rounded corners=6pt,
                   font=\sffamily\tiny\bfseries\color{anon},inner sep=2pt},

  flowa/.style={-{Stealth[length=2.4mm]},line width=0.8pt,anon},
  flown/.style={-{Stealth[length=2.4mm]},line width=0.8pt,named},

              align=center,font=\sffamily\small,inner sep=3pt},
  tier/.style={draw=studens,fill=studenslt,rounded corners=3pt,line width=0.7pt,
               align=center,font=\sffamily\small,inner sep=3pt},
  % ---------------------------------------------------------------------
  % Width-aware boxes. `wbox` takes a width and sets BOTH minimum width and
  % text width from it, so text wraps inside the box instead of overflowing.
  % Hand-placing text at guessed coordinates was the cause of every overlap in
  % the first draft of this document, so no diagram below does it any more.
  % ---------------------------------------------------------------------
  wbox/.style n args={2}{draw=line,fill=white,rounded corners=3pt,line width=0.7pt,
    align=center,inner sep=2.5pt,minimum width=#1,text width={#1-7pt},
    minimum height=#2,font=\sffamily\scriptsize},
  wtier/.style n args={2}{wbox={#1}{#2},draw=studens,fill=studenslt},
  wghost/.style n args={2}{wbox={#1}{#2},dashed,fill=paper},
  wnamed/.style n args={2}{wbox={#1}{#2},draw=named,fill=namedlt},
  wanon/.style n args={2}{wbox={#1}{#2},draw=anon,fill=anonlt},
  wfork/.style n args={2}{wbox={#1}{#2},draw=ochre,fill=ochrelt},
  % a text cell inside a wireframe: width is explicit, so columns cannot collide
  cell/.style n args={2}{anchor=north west,text width=#1,align=left,font=#2,
    inner sep=0pt,outer sep=0pt},
}
